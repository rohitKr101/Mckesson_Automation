Function oktaAuth(nextStep, Username, Password)
	wait(2)
	If nextStep = "NextWorkflowStep" or nextStep = "Approve&Follow" Then
		WshShell.SendKeys "%w"
	ElseIf nextStep = "Approve" Then
		WshShell.SendKeys "%e"
	ElseIf nextStep = "Override" Then
		WshShell.SendKeys "%o"
	End If
	
	wait(2)
	WshShell.SendKeys "%r"
	AIUtil("text_box", "Username").Type Username
'	AIUtil("button", "Next").Click
	WshShell.SendKeys "{ENTER}"
	AIUtil("text_box", "Password").Type Password
'	AIUtil("button", "Verify").Click
	WshShell.SendKeys "{ENTER}"
End Function
